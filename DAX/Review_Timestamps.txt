Review_Timestamps = 
ADDCOLUMNS (
    DISTINCT ( fact_ratings[review_timestamp] ),

    // Basic date fields
    "Review Date", DATE ( YEAR ( [review_timestamp] ), MONTH ( [review_timestamp] ), DAY ( [review_timestamp] ) ),
    "Year", YEAR ( [review_timestamp] ),
    "Month Number", MONTH ( [review_timestamp] ),
    "Month Name", FORMAT ( [review_timestamp], "MMMM" ),
    "Day", DAY ( [review_timestamp] ),
    "Day Name", FORMAT ( [review_timestamp], "DDDD" ),

    // Week number (ISO Week)
    "Week Number", WEEKNUM ( [review_timestamp], 2 ),

    // Day of week number (1 = Monday, 7 = Sunday)
    "Day of Week Number", WEEKDAY ( [review_timestamp], 2 ),

    // Day Type: Weekday or Weekend
    "Day Type",
        IF (
            WEEKDAY ( [review_timestamp], 2 ) >= 6,
            "Weekend",
            "Weekday"
        ),

    // Time components
    "Hour", HOUR ( [review_timestamp] ),
    "Minute", MINUTE ( [review_timestamp] ),

    // Phase of Day based on hour
    "Phase",
        SWITCH (
            TRUE(),
            HOUR ( [review_timestamp] ) >= 5 && HOUR ( [review_timestamp] ) < 12, "Morning",
            HOUR ( [review_timestamp] ) >= 12 && HOUR ( [review_timestamp] ) < 17, "Afternoon",
            HOUR ( [review_timestamp] ) >= 17 && HOUR ( [review_timestamp] ) < 21, "Evening",
            "Night"
        )
)
